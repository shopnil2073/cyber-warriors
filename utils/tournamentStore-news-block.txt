export const getStoredNews = (): NewsItem[] => {
  if (typeof window === "undefined") return [];
  try {
    const saved = localStorage.getItem("cw_news");
    return saved ? JSON.parse(saved) : [];
  } catch {
    return [];
  }
};

export const fetchNewsFromCloud = async (): Promise<NewsItem[]> => {
  const { data, error } = await supabase
    .from("news_articles")
    .select("*")
    .order("created_at", { ascending: false });

  if (error) {
    console.error("Supabase news read failed:", error.message);
    return getStoredNews();
  }

  const mapped = (data || []).map(mapNewsRow);
  if (typeof window !== "undefined") {
    localStorage.setItem("cw_news", JSON.stringify(mapped));
  }
  return mapped;
};

export const saveAllNews = async (newsArray: NewsItem[]): Promise<boolean> => {
  for (const article of newsArray) {
    const ok = await saveNews(article);
    if (!ok) return false;
  }
  return true;
};

export const saveNewsToCloud = saveAllNews;

export const saveNews = async (article: NewsItem): Promise<boolean> => {
  const payload = {
    title: article.title,
    slug: (article.slug || article.title).toLowerCase().trim().replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, ""),
    category: article.category || "ANNOUNCEMENT",
    excerpt: "",
    content: article.content || "",
    image_url: article.imageUrl || null,
    author_name: article.author || "CYBER WARRIORS",
    status: article.status === "Draft" ? "Draft" : "Published",
    published_at: article.status === "Draft" ? null : (article.updatedAt || new Date().toISOString()),
    updated_at: new Date().toISOString(),
  };

  const looksLikeUuid = /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(article.id);
  const result = looksLikeUuid
    ? await supabase.from("news_articles").update(payload).eq("id", article.id)
    : await supabase.from("news_articles").insert({ ...payload, legacy_id: article.id });

  if (result.error) {
    console.error("Supabase news save failed:", result.error.message);
    return false;
  }
  return true;
};

export const deleteNewsFromCloud = async (id: string): Promise<boolean> => {
  const { error } = await supabase.from("news_articles").delete().eq("id", id);
  if (error) {
    console.error("Supabase news delete failed:", error.message);
    return false;
  }
  return true;
};

export const deleteNews = deleteNewsFromCloud;
